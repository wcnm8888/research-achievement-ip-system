async page => {
  const observedRequests = [];
  const shouldTrack = (url) => {
    const lower = url.toLowerCase();
    return (
      lower.includes("/api/") ||
      lower.includes("search_logs") ||
      lower.includes("meilisearch") ||
      lower.includes("settings/config") ||
      lower.includes("attachments") ||
      lower.includes("upload") ||
      lower.includes("download") ||
      lower.includes("/fees/warnings") ||
      lower.includes("/fees")
    );
  };

  page.on("request", (request) => {
    const url = request.url();
    if (shouldTrack(url)) {
      observedRequests.push({ method: request.method(), url });
    }
  });

  await page.route("**/api/search**", async (route) => {
    await route.fulfill({
      status: 200,
      contentType: "application/json",
      body: JSON.stringify({ items: [], total: 0 }),
    });
  });

  await page.goto("http://localhost:5173/", { waitUntil: "domcontentloaded" });

  await page.evaluate(async () => {
    const mod = await import("/src/api-client.ts");
    const client = mod.createApiClient("40000000-0000-4000-8000-000000000002");
    await client.get("/search", {
      keyword: "step23b",
      targetTypes: ["ACHIEVEMENT", "FEE_RECORD"],
      take: 20,
    });
  });

  await page.waitForTimeout(250);
  await page.setViewportSize({ width: 390, height: 900 });
  await page.waitForTimeout(100);

  const searchRequests = observedRequests.filter((request) =>
    request.url.includes("/api/search"),
  );
  const parsedSearchRequests = searchRequests.map((request) => {
    const [pathWithOrigin, rawQuery = ""] = request.url.split("?");
    const path = pathWithOrigin.split("/api/")[1]
      ? `/api/${pathWithOrigin.split("/api/")[1]}`
      : pathWithOrigin;
    const queryPairs = rawQuery
      .split("&")
      .filter(Boolean)
      .map((pair) => {
        const [key, value = ""] = pair.split("=");
        return {
          key: decodeURIComponent(key),
          value: decodeURIComponent(value),
        };
      });
    const getParam = (key) =>
      queryPairs.find((pair) => pair.key === key)?.value ?? null;
    const getAllParams = (key) =>
      queryPairs.filter((pair) => pair.key === key).map((pair) => pair.value);

    return {
      method: request.method,
      path,
      keyword: getParam("keyword"),
      targetTypes: getAllParams("targetTypes"),
      take: getParam("take"),
      rawQuery: rawQuery ? `?${rawQuery}` : "",
    };
  });

  const nonGetRequests = observedRequests.filter((request) => request.method !== "GET");
  const forbiddenRequests = observedRequests.filter((request) => {
    const url = request.url.toLowerCase();
    return (
      request.method !== "GET" ||
      url.includes("search_logs") ||
      url.includes("meilisearch") ||
      url.includes("settings/config") ||
      url.includes("attachments") ||
      url.includes("upload") ||
      url.includes("download") ||
      url.includes("/fees/warnings") ||
      url.includes("/api/fees")
    );
  });

  const widthMetrics = await page.evaluate(() => ({
    body: document.body.scrollWidth,
    doc: document.documentElement.scrollWidth,
    client: document.documentElement.clientWidth,
    inner: window.innerWidth,
    overflow: document.documentElement.scrollWidth > window.innerWidth,
  }));

  const hasExpectedSearchRequest = parsedSearchRequests.some((request) =>
    request.method === "GET" &&
    request.keyword === "step23b" &&
    request.take === "20" &&
    request.targetTypes.includes("ACHIEVEMENT") &&
    request.targetTypes.includes("FEE_RECORD") &&
    request.targetTypes.length === 2
  );

  return {
    observedRequests,
    parsedSearchRequests,
    searchRequestCount: searchRequests.length,
    nonGetRequests,
    forbiddenRequests,
    widthMetrics,
    passed:
      hasExpectedSearchRequest &&
      nonGetRequests.length === 0 &&
      forbiddenRequests.length === 0 &&
      !widthMetrics.overflow,
  };
}
